public sealed class AssistantMessageRequest
{
	public string Message {get;set;}
	public Guid? ConversationId {get;set;}
}

public sealed class AssistantMessageResponse
{
    public AssistantReply Reply { get; set; } = default!;
    public AssistantWorkflow Workflow { get; set; } = default!;
}

public sealed class AssistantReply
{
    public string Kind { get; set; } = default!;
    public string Text { get; set; } = default!;
}

public sealed class AssistantWorkflow
{
    public string Status { get; set; } = default!;
    public string? Intent { get; set; }
    public string? Domain { get; set; }
    public bool AwaitingConfirmation { get; set; }
}

public sealed record TextPayload(string Text) : PresentationPayload;

public sealed record StatPayload(
    string Label,
    string Value,
    string? Unit) : PresentationPayload;

public sealed record ListPayload(
    IReadOnlyList<string> Items) : PresentationPayload;
